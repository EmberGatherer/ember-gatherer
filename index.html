import Stripe from "stripe"; const stripe = new Stripe(process.env.STRIPE_SECRET_KEY); const PACKAGES = { classic: { name: "Classic Catering", cents: 3000 }, signature: { name: "Signature Catering", cents: 5500 }, premium: { name: "Premium Catering", cents: 8000 } }; const MEATS = { whole_chicken: ["Whole Chicken", 1089], leg_quarters: ["Chicken Leg Quarters", 1089], wings: ["Chicken Wings", 1724], pork_belly: ["Pork Belly", 1270], pork_loin: ["Pork Loin", 1315], tenderloin: ["Pork Tenderloin", 1542], sirloin: ["Beef Top Sirloin", 2223], ribs: ["Pork Ribs", 1451], salmon: ["Salmon", 2676], shrimp: ["Shrimp", 2041] }; const clean = (value, max = 500) => String(value ?? "").trim().slice(0, max); export default async function handler(req, res) { if (req.method !== "POST") { res.setHeader("Allow", "POST"); return res.status(405).json({ error: "Method not allowed." }); } if (!process.env.STRIPE_SECRET_KEY) { return res.status(500).json({ error: "Stripe is not configured. Add STRIPE_SECRET_KEY in Vercel Environment Variables." }); } try { const body = req.body || {}; const name = clean(body.name, 100); const email = clean(body.email, 254).toLowerCase(); const phone = clean(body.phone, 40); const eventDate = clean(body.eventDate, 30); const location = clean(body.location, 300); const notes = clean(body.notes, 500); const packageKey = clean(body.package, 30); const guests = Number(body.guests); const selectedMeats = body.smokedMeats && typeof body.smokedMeats === "object" && !Array.isArray(body.smokedMeats) ? body.smokedMeats : {}; if (!name || !email || !phone || !eventDate || !location) { return res.status(400).json({ error: "Please complete all required contact and event fields." }); } if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { return res.status(400).json({ error: "Please enter a valid email address." }); } if (packageKey && !PACKAGES[packageKey]) { return res.status(400).json({ error: "Invalid catering package." }); } if ( packageKey && (!Number.isInteger(guests) || guests < 1 || guests > 500) ) { return res.status(400).json({ error: "Guest count must be between 1 and 500." }); } const line_items = []; if (packageKey) { const item = PACKAGES[packageKey]; line_items.push({ price_data: { currency: "cad", product_data: { name: item.name, description: "Catering package per person" }, unit_amount: item.cents }, quantity: guests }); } const meatDescriptions = []; for (const [id, rawQty] of Object.entries(selectedMeats)) { if (!MEATS[id]) { return res.status(400).json({ error: "Unknown smoked meat item." }); } const quantity = Number(rawQty); if ( !Number.isInteger(quantity) || quantity < 0 || quantity > 200 ) { return res.status(400).json({ error: "Smoked meat quantities must be whole pounds from 0 to 200." }); } if (quantity > 0) { const [productName, unitAmount] = MEATS[id]; line_items.push({ price_data: { currency: "cad", product_data: { name: productName, description: "Finished product, per pound" }, unit_amount: unitAmount }, quantity: quantity }); meatDescriptions.push( `${productName}: ${quantity} lb` ); } } if (!line_items.length) { return res.status(400).json({ error: "Choose a catering package or at least one smoked meat item." }); } const baseUrl = ( process.env.SITE_URL || `https://${req.headers.host}` ).replace(/\/$/, ""); const metadata = { customer_name: name, customer_phone: phone, event_date: eventDate, event_location: location, guest_count: String(guests || 0), catering_package: packageKey || "Smoked meats only", smoked_meats: meatDescriptions.join("; ").slice(0, 500), menu_notes: notes }; const session = await stripe.checkout.sessions.create({ mode: "payment", payment_method_types: ["card"], line_items, customer_email: email, billing_address_collection: "auto", phone_number_collection: { enabled: true }, metadata, payment_intent_data: { metadata }, success_url: `${baseUrl}/?payment=success&session_id={CHECKOUT_SESSION_ID}`, cancel_url: `${baseUrl}/?payment=cancelled` }); return res.status(200).json({ url: session.url }); } catch (error) { console.error("Stripe Checkout error:", error); return res.status(500).json({ error: "Unable to create Stripe checkout. Please try again or contact Ember & Gatherer." }); } }
